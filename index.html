                  function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const products = readSheet(ss, 'Products', ['code','description','quantity','buyingPrice','sellingPrice']);
  const billsRaw = readSheet(ss, 'Bills', ['billNo','timestamp','customerMobile','paymentMode','amountReceived','changeReturned','total']);
  const itemsRaw = readSheet(ss, 'BillItems', ['billNo','code','description','qty','sellingPrice','buyingPrice']);
  const settings = readSettings(ss);

  const bills = billsRaw.map(function(b) {
    const items = itemsRaw.filter(function(it){ return String(it.billNo) === String(b.billNo); })
      .map(function(it){
        return {code: it.code, description: it.description, qty: Number(it.qty), sellingPrice: Number(it.sellingPrice), buyingPrice: Number(it.buyingPrice)};
      });
    return {
      billNo: Number(b.billNo),
      timestamp: b.timestamp,
      customerMobile: b.customerMobile || '',
      paymentMode: b.paymentMode || 'cash',
      amountReceived: b.amountReceived === '' ? null : Number(b.amountReceived),
      changeReturned: b.changeReturned === '' ? null : Number(b.changeReturned),
      total: Number(b.total),
      items: items
    };
  });

  const payload = {
    products: products.map(function(p){
      return {code: p.code, description: p.description, quantity: Number(p.quantity), buyingPrice: Number(p.buyingPrice), sellingPrice: Number(p.sellingPrice)};
    }),
    bills: bills,
    shopDetails: {name: settings.ShopName || '', address: settings.ShopAddress || '', phone: settings.ShopPhone || '', logoUrl: settings.ShopLogoUrl || ''},
    nextBillNo: Number(settings.NextBillNo || 1001)
  };

  return ContentService.createTextOutput(JSON.stringify(payload)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  writeSheet(ss, 'Products', ['code','description','quantity','buyingPrice','sellingPrice'],
    (data.products || []).map(function(p){ return [p.code, p.description, p.quantity, p.buyingPrice, p.sellingPrice]; }));

  const billRows = [];
  const itemRows = [];
  (data.bills || []).forEach(function(b){
    billRows.push([b.billNo, b.timestamp, b.customerMobile || '', b.paymentMode || 'cash',
      (b.amountReceived == null ? '' : b.amountReceived), (b.changeReturned == null ? '' : b.changeReturned), b.total]);
    (b.items || []).forEach(function(it){
      itemRows.push([b.billNo, it.code, it.description, it.qty, it.sellingPrice, it.buyingPrice]);
    });
  });
  writeSheet(ss, 'Bills', ['billNo','timestamp','customerMobile','paymentMode','amountReceived','changeReturned','total'], billRows);
  writeSheet(ss, 'BillItems', ['billNo','code','description','qty','sellingPrice','buyingPrice'], itemRows);

  writeSettings(ss, {
    ShopName: data.shopDetails ? data.shopDetails.name : '',
    ShopAddress: data.shopDetails ? data.shopDetails.address : '',
    ShopPhone: data.shopDetails ? data.shopDetails.phone : '',
    ShopLogoUrl: data.shopDetails ? data.shopDetails.logoUrl : '',
    NextBillNo: data.nextBillNo || 1001
  });

  return ContentService.createTextOutput(JSON.stringify({ok: true})).setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return sheet;
}

function readSheet(ss, name, headers) {
  const sheet = ss.getSheetByName(name);
  if (!sheet) return [];
  const values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  const rows = values.slice(1);
  return rows.filter(function(r){ return r.join('') !== ''; }).map(function(r){
    const obj = {};
    headers.forEach(function(h,i){ obj[h] = r[i]; });
    return obj;
  });
}

function writeSheet(ss, name, headers, rows) {
  const sheet = getOrCreateSheet(ss, name, headers);
  sheet.clearContents();
  sheet.appendRow(headers);
  if (rows.length) sheet.getRange(2,1,rows.length, headers.length).setValues(rows);
}

function readSettings(ss) {
  const sheet = ss.getSheetByName('Settings');
  if (!sheet) return {};
  const values = sheet.getDataRange().getValues();
  const out = {};
  values.forEach(function(r){ if (r[0]) out[r[0]] = r[1]; });
  return out;
}

function writeSettings(ss, obj) {
  const sheet = getOrCreateSheet(ss, 'Settings', ['Key','Value']);
  sheet.clearContents();
  sheet.appendRow(['Key','Value']);
  const rows = Object.keys(obj).map(function(k){ return [k, obj[k]]; });
  if (rows.length) sheet.getRange(2,1,rows.length,2).setValues(rows);
}
